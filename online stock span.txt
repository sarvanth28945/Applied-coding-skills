class StockSpanner {
    private int[][] stack;
    private int top;

    public StockSpanner() {
        stack = new int[10000][2];
        top = -1;
    }
    
    public int next(int price) {
        int span = 1;
        while (top >= 0 && stack[top][0] <= price) {
            span += stack[top][1];
            top--;
        }
        top++;
        stack[top][0] = price;
        stack[top][1] = span;
        return span;
    }
}
